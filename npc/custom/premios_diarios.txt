prontera,5,5,5	script	Recompensa_Login	111,{

OnPCLoginEvent:
	attachrid getcharid(3);

	if (!@logintime)
		@logintime = gettimetick(2);

	callsub S_CheckReward;
	end;

OnLoginCmnd:
	attachrid getcharid(3);
	callsub S_CheckReward;
	end;

S_CheckReward:

	.@hoje = gettime(DT_YEAR) * 12 * 31 + gettime(DT_MONTH) * 31 + gettime(DT_DAYOFMONTH);

	if (.Reset && .@hoje > #LastDailyReward + 1)
		#DRewardCon = 0;
	if (.Reset && .@hoje > #LastVIPReward + 1)
		#VIPRewardCon = 0;

	.@vipSize = getarraysize(.VIPRewards$);
	.@size = getarraysize(.Rewards$);

	if (#DRewardCon >= .@size && #VIPRewardCon >= .@vipSize) {
		#LastDailyReward = .@hoje;
		#LastVIPReward = .@hoje;
		return;
	}

	sleep2 1000;

	for (.@k = 0; .@k < 2; .@k++) {

		if (!.@k) {
			if (!vip_status(VIP_STATUS_ACTIVE))
				continue;
			if (#VIPRewardCon >= .@vipSize)
				continue;
			if (.@hoje <= #LastVIPReward)
				continue;

			explode(.@XT$,.VIPRewards$[#VIPRewardCon+1],",");

		} else {
			if (#DRewardCon >= .@size)
				continue;
			if (.@hoje <= #LastDailyReward)
				continue;

			explode(.@XT$,.Rewards$[#DRewardCon+1],",");
		}

		deletearray .@itms[0], getarraysize(.@itms);
		deletearray .@qnts[0], getarraysize(.@qnts);

		.@y = 0;
		for (.@i = 4; .@i < getarraysize(.@XT$); .@i += 2) {
		.@itms[.@y] = atoi(.@XT$[.@i]);
		.@qnts[.@y] = atoi(.@XT$[.@i+1]);
		.@y++;
		}

		if (!checkweight2(.@itms,.@qnts)) {
			message strcharinfo(0),"[Recompensa]: Peso excessivo. Guarde itens e relogue.";
			return;
		}

		for (.@i = 0; .@i < .@y; .@i++) {
			getitem .@itms[.@i], .@qnts[.@i];
		}

		if (atoi(.@XT$[0]) > 0) {
			zeny += atoi(.@XT$[0]);
			message strcharinfo(0),"[Recompensa]: Você recebeu "+atoi(.@XT$[0])+" Zeny.";
		}

		if (!.@k) {
			#VIPRewardCon++;
			#LastVIPReward = .@hoje;
			message strcharinfo(0),"[Recompensa]: Recompensa VIP recebida.";
		} else {
			#DRewardCon++;
			#LastDailyReward = .@hoje;
			message strcharinfo(0),"[Recompensa]: Recompensa diária recebida.";
		}
	}
	return;

OnNextCmnd:
	attachrid getcharid(3);
	message strcharinfo(0),"[Recompensa]: Use @collectdaily para coletar.";
	end;

OnHour00:
	if (gettime(DT_DAYOFMONTH) == 1) {
		query_sql("DELETE FROM `acc_reg_num` WHERE `key` IN ('#DRewardCon','#VIPRewardCon','#LastVIPReward','#LastDailyReward')");
	}
	end;

OnInit:
	.Reset = 1;

	bindatcmd "collectdaily", strnpcinfo(3)+"::OnLoginCmnd", 0, 99;
	bindatcmd "nextreward", strnpcinfo(3)+"::OnNextCmnd", 0, 99;

	setarray .Rewards$[1],
		"1000",
		"0,0,0,0,501,5",
		"0,0,0,0,506,5",
		"2000",
		"2000",
		"0,0,0,0,502,5",
		"0,0,0,0,12208,1",
		"2500";

	setarray .VIPRewards$[1],
		"2000",
		"0,0,0,0,501,10",
		"0,0,0,0,506,10",
		"4000",
		"4000",
		"0,0,0,0,502,10",
		"0,0,0,0,12208,2",
		"5000";
	end;
}
