//===========================================================
// NPC Recompensa Diária + VIP
// Compatível: rAthena
//===========================================================

prontera,150,180,5	script	Recompensa Diária	4_F_KAFRA1,{

	mes "[Recompensa]";
	mes "Olá, ^0000FF"+strcharinfo(0)+"^000000!";
	mes "O que deseja fazer?";
	next;

	switch(select("Coletar recompensa:Ver próxima recompensa:Sair")) {

	case 1:
		callsub S_CheckReward;
		close;

	case 2:
		callsub S_ShowNext;
		close;

	default:
		close;
	}

//===========================================================
// ENTREGA RECOMPENSA
//===========================================================
S_CheckReward:

	.@today = gettime(DT_YEAR)*10000 + gettime(DT_MONTH)*100 + gettime(DT_DAYOFMONTH);

	if (.Reset && .@today > #LastDailyReward + 1)
		#DRewardCon = 0;
	if (.Reset && .@today > #LastVIPReward + 1)
		#VIPRewardCon = 0;

	// DAILY
	if (.@today > #LastDailyReward) {
		if (#DRewardCon < getarraysize(.Rewards$)) {
			callsub S_GiveReward, .Rewards$[#DRewardCon+1];
			#DRewardCon++;
			#LastDailyReward = .@today;
			mes "[Recompensa]";
			mes "Você recebeu sua ^0000FFrecompensa diária^000000!";
		} else {
			mes "[Recompensa]";
			mes "Você já completou todas as recompensas diárias.";
		}
	} else {
		mes "[Recompensa]";
		mes "Você já coletou a recompensa diária hoje.";
	}

	// VIP
	if (vip_status(VIP_STATUS_ACTIVE)) {
		if (.@today > #LastVIPReward) {
			if (#VIPRewardCon < getarraysize(.VIPRewards$)) {
				callsub S_GiveReward, .VIPRewards$[#VIPRewardCon+1];
				#VIPRewardCon++;
				#LastVIPReward = .@today;
				mes "E também sua ^0000FFrecompensa VIP^000000!";
			}
		}
	}

	close;

//===========================================================
// MOSTRAR PRÓXIMA
//===========================================================
S_ShowNext:

	mes "[Recompensa]";

	if (#DRewardCon >= getarraysize(.Rewards$)) {
		mes "Você já completou todas as recompensas diárias.";
		close;
	}

	.@data$ = .Rewards$[#DRewardCon+1];
	explode(.@X$, .@data$, ",");

	if (atoi(.@X$[0]) > 0)
		mes "Zeny: "+atoi(.@X$[0]);

	for (.@i = 4; .@i < getarraysize(.@X$); .@i += 2)
		mes "Item "+getitemname(atoi(.@X$[.@i]))+" x"+atoi(.@X$[.@i+1]);

	close;

//===========================================================
// FUNÇÃO ENTREGA
//===========================================================
S_GiveReward:
	.@data$ = getarg(0);
	explode(.@X$, .@data$, ",");

	deletearray .@itms;
	deletearray .@qnts;

	.@y = 0;
	for (.@i = 4; .@i < getarraysize(.@X$); .@i += 2) {
		.@itms[.@y] = atoi(.@X$[.@i]);
		.@qnts[.@y] = atoi(.@X$[.@i+1]);
		.@y++;
	}

	if (!checkweight2(.@itms, .@qnts)) {
		mes "[Recompensa]";
		mes "Peso excessivo. Libere espaço e tente novamente.";
		close;
	}

	for (.@i = 0; .@i < .@y; .@i++)
		getitem .@itms[.@i], .@qnts[.@i];

	if (atoi(.@X$[0]) > 0)
		zeny += atoi(.@X$[0]);

	return;

//===========================================================
// INIT
//===========================================================
OnInit:
	.Reset = 1;

	setarray .Rewards$[1],
		"1000",
		"0,0,0,0,501,5",
		"0,0,0,0,506,5",
		"2000",
		"2000",
		"0,0,0,0,502,5",
		"0,0,0,0,12208,1",
		"2500";

	setarray .VIPRewards$[1],
		"2000",
		"0,0,0,0,501,10",
		"0,0,0,0,506,10",
		"4000",
		"4000",
		"0,0,0,0,502,10",
		"0,0,0,0,12208,2",
		"5000";
	end;
}
